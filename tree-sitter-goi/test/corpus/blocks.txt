==================

	tickets: [ticket]
	onOpen: func(id: string, count: number)
}

type ticket { id: string, title: string }

Window {
	Column {
		{#each bridge.tickets as ticket}
			Button { label: ticket.title onClick: bridge.onOpen(ticket.id, 2) }
		{/each}
	}
}
